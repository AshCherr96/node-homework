# Task 1: Find the total price of each of the first 5 orders, grouped by order_id
SELECT o.order_id, SUM(p.price * li.quantity) AS total_price 
FROM orders o 
JOIN line_items li ON o.order_id = li.order_id 
JOIN products p ON li.product_id = p.product_id 
GROUP BY o.order_id 
ORDER BY o.order_id 
LIMIT 5;

# Task 2: Find the average total price of each customer's orders, ordered by customer name using a subquery
SELECT c.customer_name, AVG(t.total_price) AS average_order_price
FROM customers c
JOIN (
  SELECT o.customer_id, SUM(p.price * li.quantity) AS total_price
  FROM orders o
  JOIN line_items li ON o.order_id = li.order_id
  JOIN products p ON li.product_id = p.product_id
  GROUP BY o.order_id, o.customer_id
) AS t ON c.customer_id = t.customer_id
GROUP BY c.customer_id, c.customer_name
ORDER BY c.customer_name;

SELECT customer_id FROM customers WHERE customer_name = 'Perez and Sons';

SELECT employee_id FROM employees WHERE first_name = 'Miranda' AND last_name = 'Harris';

SELECT product_id FROM products ORDER BY price ASC LIMIT 5;

# Task 3: Transaction to insert order and insert 5 line items using the returned order_id
BEGIN;
WITH new_order AS (
  INSERT INTO orders (customer_id, employee_id, date) 
  VALUES (
    (SELECT customer_id FROM customers WHERE customer_name = 'Perez and Sons'), 
    (SELECT employee_id FROM employees WHERE first_name = 'Miranda' AND last_name = 'Harris'), 
    '2025-03-11'
  ) 
  RETURNING order_id
)
INSERT INTO line_items (order_id, product_id, quantity) 
SELECT new_order.order_id, p.product_id, 10
FROM new_order, (
  SELECT product_id FROM products ORDER BY price ASC LIMIT 5
) AS p;
COMMIT;

# Task 3: Select all line items corresponding to the newly inserted order
SELECT * FROM line_items WHERE order_id = (
  SELECT o.order_id FROM orders o
  JOIN customers c ON o.customer_id = c.customer_id
  JOIN employees e ON o.employee_id = e.employee_id
  WHERE c.customer_name = 'Perez and Sons' 
    AND e.first_name = 'Miranda' 
    AND e.last_name = 'Harris'
  ORDER BY o.order_id DESC 
  LIMIT 1
);

# Task 4: Find employees with more than five orders using HAVING
SELECT e.first_name, e.last_name, COUNT(o.order_id) AS order_count 
FROM employees e 
JOIN orders o ON e.employee_id = o.employee_id 
GROUP BY e.employee_id, e.first_name, e.last_name 
HAVING COUNT(o.order_id) > 5 
ORDER BY e.last_name;
